Example 01

Counter with Time-Travel

Fine-grained signals, computed values, and enableTimeTravel() for undo/redo. Demonstrates the core reactive primitives.

import {
  signal, computed, effect,
  enableTimeTravel, h, mountApp, show,
} from '@nexoraaidrishti/runtime';

// Enable time travel before creating signals
const tt = enableTimeTravel({ maxEntries: 50 });

const count   = signal(0);
const doubled = computed(() => count() * 2);
const squared = computed(() => count() ** 2);
const isEven  = computed(() => count() % 2 === 0);

// Log every change (effect cleans up on re-run via return)
effect(() => {
  console.log(`count changed → ${count()}`);
});

function CounterApp() {
  return h('div', { class: 'counter-app' },
    // Display row
    h('div', { class: 'stats' },
      h('div', { class: 'stat' },
        h('span', { class: 'stat-value' }, () => count()),
        h('span', { class: 'stat-label' }, 'Count'),
      ),
      h('div', { class: 'stat' },
        h('span', { class: 'stat-value' }, () => doubled()),
        h('span', { class: 'stat-label' }, 'Doubled'),
      ),
      h('div', { class: 'stat' },
        h('span', { class: 'stat-value' }, () => squared()),
        h('span', { class: 'stat-label' }, 'Squared'),
      ),
    ),

    // Parity badge — reactive class + text
    h('span', {
      class: () => isEven() ? 'badge badge-green' : 'badge'
    }, () => isEven() ? 'Even' : 'Odd'),

    // Controls
    h('div', { class: 'controls' },
      h('button', { class: 'btn btn-secondary', onclick: () => count.set(c => c - 1) }, '−'),
      h('button', { class: 'btn btn-primary',   onclick: () => count.set(c => c + 1) }, '+'),
      h('button', { class: 'btn btn-ghost',     onclick: () => count.set(0) }, 'Reset'),
    ),

    // Time-travel controls
    h('div', { class: 'tt-bar' },
      h('button', {
        class: 'btn btn-ghost btn-sm',
        onclick: () => tt.travel(-1),
        disabled: () => tt.present() === 0,
      }, '← Undo'),
      h('span', { class: 'text-muted' },
        () => `Step ${tt.present()} / ${tt.entries().length}`
      ),
      h('button', {
        class: 'btn btn-ghost btn-sm',
        onclick: () => tt.travel(+1),
        disabled: () => tt.present() === tt.entries().length,
      }, 'Redo →'),
    ),
  );
}

mountApp(CounterApp, document.getElementById('app')!);
Example 02

TodoMVC

Full TodoMVC implementation with filters, add/complete/clear, and reactive computed counts. Shows each(), show(), and batch().

import { signal, computed, h, each, show, batch, mountApp } from '@nexoraaidrishti/runtime';

interface Todo { id: number; text: string; done: boolean; }
type Filter = 'all' | 'active' | 'completed';

const todos   = signal<Todo[]>([]);
const input   = signal('');
const filter  = signal<Filter>('all');

const visible   = computed(() => {
  const f = filter();
  return todos().filter(t =>
    f === 'all'       ? true :
    f === 'active'    ? !t.done :
    t.done
  );
});
const remaining  = computed(() => todos().filter(t => !t.done).length);
const allDone    = computed(() => todos().length > 0 && remaining() === 0);

const addTodo = () => {
  const text = input().trim();
  if (!text) return;
  todos.set([...todos(), { id: Date.now(), text, done: false }]);
  input.set('');
};

const toggle = (id: number) =>
  todos.set(todos().map(t => t.id === id ? { ...t, done: !t.done } : t));

const remove = (id: number) =>
  todos.set(todos().filter(t => t.id !== id));

const clearCompleted = () =>
  todos.set(todos().filter(t => !t.done));

const toggleAll = () =>
  todos.set(todos().map(t => ({ ...t, done: !allDone() })));

function TodoApp() {
  return h('section', { class: 'todoapp' },
    h('header', {},
      h('h1', {}, 'todos'),
      h('input', {
        class: 'new-todo',
        placeholder: 'What needs to be done?',
        value: input,
        oninput: (e) => input.set(e.target.value),
        onkeydown: (e) => e.key === 'Enter' && addTodo(),
      }),
    ),
    show(() => todos().length > 0,
      () => h('section', { class: 'main' },
        h('input', {
          id: 'toggle-all', type: 'checkbox',
          checked: allDone, onchange: toggleAll,
        }),
        each(() => visible(), t => t.id, (t) =>
          h('li', { class: () => t.done ? 'completed' : '' },
            h('input', { type: 'checkbox', checked: t.done,
              onchange: () => toggle(t.id) }),
            h('label', {}, t.text),
            h('button', { class: 'destroy', onclick: () => remove(t.id) }, '×'),
          )
        ),
      )
    ),
    show(() => todos().length > 0,
      () => h('footer', { class: 'footer' },
        h('span', {}, () => `${remaining()} item${remaining() === 1 ? '' : 's'} left`),
        h('ul', { class: 'filters' },
          ...['all', 'active', 'completed'].map(f =>
            h('li', {},
              h('a', {
                class: () => filter() === f ? 'selected' : '',
                onclick: () => filter.set(f as Filter),
              }, f[0].toUpperCase() + f.slice(1))
            )
          )
        ),
        show(() => todos().some(t => t.done),
          () => h('button', { onclick: clearCompleted }, 'Clear completed')
        ),
      )
    ),
  );
}

mountApp(TodoApp, document.getElementById('app')!);

Data Fetching

createResource with reactive loading, error, and retry states. Skeleton UI while fetching, error recovery, and dependent queries.

import { signal, createResource, createQuery, h, show, each, mountApp } from '@nexoraaidrishti/runtime';

interface Post { id: number; title: string; body: string; userId: number; }
interface User { id: number; name: string; email: string; }

const selectedId = signal<number | null>(null);
const page       = signal(1);

// Posts re-fetch automatically when page changes
const posts = createResource<Post[]>(
  () => fetch(`https://jsonplaceholder.typicode.com/posts?_page=${page()}&_limit=5`)
          .then(r => r.json()),
  { initialValue: [] }
);

// User resource — only fetches when selectedId is non-null
const selectedUser = createResource<User | null>(
  () => selectedId()
    ? fetch(`https://jsonplaceholder.typicode.com/users/${selectedId()}`)
        .then(r => r.json())
    : Promise.resolve(null),
  { initialValue: null }
);

function Skeleton() {
  return h('div', { class: 'skeleton-list' },
    ...Array(5).fill(0).map(() =>
      h('div', { class: 'skeleton-item',
        style: 'height:60px;background:var(--bg-hover);border-radius:8px;margin:8px 0;animation:pulse 1.5s infinite'
      })
    )
  );
}

function DataApp() {
  return h('div', { class: 'data-app' },

    show(() => !!posts.error(),
      () => h('div', { class: 'callout callout-warn' },
        'Error loading posts. ',
        h('button', { onclick: () => posts.refetch() }, 'Retry'),
      )
    ),

    show(() => posts.loading(),
      Skeleton,
      () => each(() => posts.data(), p => p.id, (post) =>
        h('div', {
          class: () => selectedId() === post.userId ? 'post-card active' : 'post-card',
          onclick: () => selectedId.set(post.userId),
        },
          h('h3', {}, post.title),
          h('p', {}, post.body.slice(0, 80) + '...'),
        )
      )
    ),

    h('div', { class: 'pagination' },
      h('button', { onclick: () => page.set(p => Math.max(1, p - 1)), disabled: () => page() === 1 }, 'Previous'),
      () => `Page ${page()}`,
      h('button', { onclick: () => page.set(p => p + 1) }, 'Next'),
    ),
  );
}

mountApp(DataApp, document.getElementById('app')!);

Form with Validation

createForm with schema-driven validation, field-level errors, touched tracking, and async submission handling.

import { createForm, signal, h, show, mountApp } from '@nexoraaidrishti/runtime';

const submitStatus = signal<'idle' | 'submitting' | 'success' | 'error'>('idle');

const form = createForm({
  name:     { initial: '',    validate: v => v.length >= 2  || 'Name must be at least 2 characters' },
  email:    { initial: '',    validate: v => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v) || 'Invalid email address' },
  password: { initial: '',    validate: v => v.length >= 8  || 'Password must be at least 8 characters' },
  age:      { initial: '',    validate: v => Number(v) >= 18 || 'Must be 18 or older' },
  terms:    { initial: false, validate: v => v === true     || 'You must accept the terms' },
}, {
  onSubmit: async (values) => {
    submitStatus.set('submitting');
    try {
      await fetch('/api/register', {
        method: 'POST',
        body: JSON.stringify(values),
        headers: { 'Content-Type': 'application/json' },
      });
      submitStatus.set('success');
    } catch {
      submitStatus.set('error');
    }
  }
});

function Field({ label, field, type = 'text' }: { label: string; field: any; type?: string }) {
  return h('div', { class: 'field' },
    h('label', {}, label),
    h('input', {
      type,
      value: field.value,
      oninput: (e: any) => field.value.set(e.target.value),
      onblur:  () => field.touch(),
      class: () => field.error() && field.touched() ? 'invalid' : '',
    }),
    show(() => !!(field.error() && field.touched()),
      () => h('p', { class: 'field-error text-red' }, () => field.error())
    ),
  );
}

function RegisterForm() {
  return show(() => submitStatus() === 'success',
    () => h('div', { class: 'callout callout-tip' }, 'Registration successful!'),
    () => h('form', { onsubmit: (e) => { e.preventDefault(); form.submit(); } },
      Field({ label: 'Full Name', field: form.fields.name }),
      Field({ label: 'Email',     field: form.fields.email,    type: 'email' }),
      Field({ label: 'Password',  field: form.fields.password, type: 'password' }),
      Field({ label: 'Age',       field: form.fields.age,      type: 'number' }),
      h('button', {
        type: 'submit',
        class: 'btn btn-primary',
        disabled: () => !form.valid() || submitStatus() === 'submitting',
      }, () => submitStatus() === 'submitting' ? 'Submitting...' : 'Create Account'),
    )
  );
}

mountApp(RegisterForm, document.getElementById('app')!);

Wizard State Machine

createMachine with guarded transitions for a multi-step form. Guards prevent advancing without valid data. Entry hooks log analytics.

import { createMachine, signal, computed, h, show, mountApp } from '@nexoraaidrishti/runtime';

// Step data
const name    = signal('');
const email   = signal('');
const plan    = signal<'free' | 'pro' | 'team'>('free');
const payment = signal('');

// Guards — each step validates before advancing
const step1Valid = computed(() => name().length >= 2);
const step2Valid = computed(() => /^[^\s@]+@[^\s@]+/.test(email()));
const step3Valid = computed(() => plan() === 'free' || payment().length === 16);

const wizard = createMachine({
  initial: 'name',
  states: {
    name: {
      on: { NEXT: { target: 'email',   guard: () => step1Valid() } },
      entry: () => trackStep('name'),
    },
    email: {
      on: {
        NEXT: { target: 'plan', guard: () => step2Valid() },
        BACK: 'name',
      },
      entry: () => trackStep('email'),
    },
    plan: {
      on: {
        NEXT: { target: 'payment', guard: () => step3Valid() },
        BACK: 'email',
      },
    },
    payment: {
      on: { NEXT: 'confirm', BACK: 'plan' }
    },
    confirm: {},
  },
});

const steps = ['name', 'email', 'plan', 'payment', 'confirm'];
const stepIndex = computed(() => steps.indexOf(wizard.state()));
const trackStep = (s: string) => console.log('wizard:step:' + s);

function WizardApp() {
  return h('div', { class: 'wizard' },
    // Progress bar
    h('div', { class: 'progress' },
      h('div', {
        class: 'progress-bar',
        style: () => `width:${((stepIndex() + 1) / steps.length) * 100}%`,
      })
    ),
    () => `Step ${stepIndex() + 1} of ${steps.length}: ${wizard.state()}`,

    // Step 1 — Name
    show(() => wizard.state() === 'name', () =>
      h('div', {},
        h('input', { placeholder: 'Your name', value: name, oninput: (e) => name.set(e.target.value) }),
      )
    ),
    // Step 2 — Email
    show(() => wizard.state() === 'email', () =>
      h('input', { type: 'email', placeholder: 'Email address', value: email, oninput: (e) => email.set(e.target.value) })
    ),
    // Navigation buttons
    h('div', { class: 'wizard-nav' },
      show(() => wizard.can('BACK'), () =>
        h('button', { class: 'btn btn-ghost', onclick: () => wizard.send('BACK') }, 'Back')
      ),
      show(() => wizard.state() !== 'confirm', () =>
        h('button', {
          class: 'btn btn-primary',
          onclick: () => wizard.send('NEXT'),
          disabled: () => !wizard.can('NEXT'),
        }, 'Next →')
      ),
    ),
  );
}

mountApp(WizardApp, document.getElementById('app')!);

LLM Chat Interface

createLLMStream with streaming tokens, thinking indicator, tool call state, and message history. Full chat UI in under 60 lines.

import {
  signal, computed,
  createLLMStream, createThinkingState, createToolCallState,
  h, show, each, mountApp,
} from '@nexoraaidrishti/runtime';

interface Message { role: 'user' | 'assistant'; content: string; }

const messages  = signal<Message[]>([]);
const userInput = signal('');
const thinking  = createThinkingState();
const toolCalls = createToolCallState();
const currentTokens = signal(0);

const stream = createLLMStream({
  endpoint: '/api/chat',
  onStart: () => {
    currentTokens.set(0);
    messages.set([...messages(), { role: 'assistant', content: '' }]);
  },
  onChunk: (chunk) => {
    const list = messages();
    const last = list[list.length - 1];
    messages.set([...list.slice(0, -1), { ...last, content: last.content + chunk }]);
    currentTokens.set(currentTokens() + 1);
  },
  onDone: () => thinking.stop(),
  onError: (err) => console.error('Stream error:', err),
  onTool: (call) => toolCalls.add(call),
});

const sendMessage = async () => {
  const text = userInput().trim();
  if (!text || stream.status() === 'streaming') return;

  messages.set([...messages(), { role: 'user', content: text }]);
  userInput.set('');
  thinking.start();

  await stream.start({
    messages: messages().map(m => ({ role: m.role, content: m.content })),
  });
};

function ChatApp() {
  return h('div', { class: 'chat-app' },
    h('div', { class: 'messages' },
      each(() => messages(), (_, i) => i, (msg) =>
        h('div', { class: `message message-${msg.role}` },
          h('strong', {}, msg.role === 'user' ? 'You' : 'AI'),
          h('p', {}, msg.content),
        )
      ),
      show(() => thinking.active(), () =>
        h('div', { class: 'thinking-indicator' },
          h('span', {}, 'AI is thinking'),
          h('span', { class: 'dots' }, '...'),
          () => `${currentTokens()} tokens`,
        )
      ),
    ),
    h('div', { class: 'chat-input' },
      h('textarea', {
        value: userInput,
        oninput: (e) => userInput.set(e.target.value),
        onkeydown: (e) => e.key === 'Enter' && !e.shiftKey && (e.preventDefault(), sendMessage()),
        placeholder: 'Ask anything... (Enter to send, Shift+Enter for newline)',
      }),
      h('button', {
        class: 'btn btn-primary',
        onclick: sendMessage,
        disabled: () => stream.status() === 'streaming',
      }, () => stream.status() === 'streaming' ? 'Streaming...' : 'Send'),
      show(() => stream.status() === 'streaming', () =>
        h('button', { class: 'btn btn-ghost', onclick: () => stream.abort() }, 'Stop')
      ),
    ),
  );
}

mountApp(ChatApp, document.getElementById('app')!);

Feature Flag A/B Test

createFlag for boolean and variant flags, createBandit for epsilon-greedy traffic allocation, and reactive UI that updates when flag values change.

import { createFlag, createBandit, signal, computed, h, show, mountApp } from '@nexoraaidrishti/runtime';

// Simple boolean flag
const newUiFlag = createFlag('new-ui', {
  defaultValue: false,
  targeting: { percentile: 50 },  // 50% of users
});

// Multi-variant flag
const ctaFlag = createFlag<'control' | 'variant-a' | 'variant-b'>('cta-text', {
  defaultValue: 'control',
  variants: {
    'control':   { weight: 34 },
    'variant-a': { weight: 33 },
    'variant-b': { weight: 33 },
  },
});

// Multi-armed bandit — automatically shifts traffic to the winner
const bandit = createBandit({
  arms: ['control', 'variant-a', 'variant-b'],
  algorithm: 'epsilon-greedy',
  epsilon: 0.1,
  onConvert: (arm, reward) => console.log(`Arm ${arm} converted with reward ${reward}`),
});

// Compute the CTA text for the current variant
const ctaText = computed(() => {
  const variant = ctaFlag.value();
  return {
    'control':   'Sign Up Free',
    'variant-a': 'Start Building Today',
    'variant-b': 'Get Started in 60 Seconds',
  }[variant] ?? 'Sign Up';
});

const clicks     = signal(0);
const conversions = signal(0);
const ctr         = computed(() => clicks() === 0 ? '0%' : `${((conversions() / clicks()) * 100).toFixed(1)}%`);

function FlagDemo() {
  return h('div', { class: 'flag-demo' },

    show(() => newUiFlag.value(),
      () => h('div', { class: 'callout callout-tip' }, 'New UI is enabled for you!'),
      () => h('div', { class: 'callout callout-info' }, 'You are in the control group.'),
    ),

    h('div', { class: 'ab-hero' },
      h('h2', {}, () => ctaText()),
      h('p', { class: 'text-muted' }, () => `Variant: ${ctaFlag.value()}`),
      h('button', {
        class: 'btn btn-primary btn-lg',
        onclick: () => {
          clicks.set(c => c + 1);
          if (Math.random() > 0.7) {
            conversions.set(c => c + 1);
            bandit.reward(ctaFlag.value(), 1);
          }
        },
      }, () => ctaText()),
    ),

    h('div', { class: 'stats' },
      h('div', { class: 'stat' }, () => `Clicks: ${clicks()}`),
      h('div', { class: 'stat' }, () => `Conversions: ${conversions()}`),
      h('div', { class: 'stat' }, () => `CTR: ${ctr()}`),
      h('div', { class: 'stat' }, () => `Bandit arm: ${bandit.bestArm()}`),
    ),
  );
}

mountApp(FlagDemo, document.getElementById('app')!);

Real-time Collaboration

createCollabSession with CRDT-backed signal syncing, live presence indicators, and cursor positions. Shows the full multi-user reactive model.

import {
  createCollabSession, signal, computed,
  h, show, each, mountApp,
} from '@nexoraaidrishti/runtime';

const COLORS = ['#6366f1', '#10b981', '#f59e0b', '#ef4444', '#3b82f6'];

// Each user gets a random ID and color
const myId    = 'user-' + Math.random().toString(36).slice(2, 8);
const myColor = COLORS[Math.floor(Math.random() * COLORS.length)];

const session = createCollabSession({
  room:     'drishti-demo-doc',
  userId:   myId,
  wsUrl:    'wss://collab.drishti.dev',
  presence: true,
  meta:     { color: myColor, name: 'Anonymous ' + myId.slice(-4) },
});

// These signals are synced via CRDT across all connected clients
const title   = signal('Untitled Document');
const content = signal('');
const tags    = signal<string[]>([]);

session.sync('title',   title);
session.sync('content', content);
session.sync('tags',    tags);

// Derived reactive values
const peerCount  = computed(() => session.peers().length);
const wordCount  = computed(() => content().split(/\s+/).filter(Boolean).length);
const charCount  = computed(() => content().length);
const newTag     = signal('');

const addTag = () => {
  const t = newTag().trim();
  if (t && !tags().includes(t)) tags.set([...tags(), t]);
  newTag.set('');
};

function CollabDoc() {
  return h('div', { class: 'collab-doc' },

    // Presence bar
    h('div', { class: 'presence-bar' },
      h('div', { class: 'presence-avatars' },
        // My avatar
        h('span', {
          class: 'avatar',
          style: `background:${myColor};outline:2px solid ${myColor}`,
          title: 'You (me)',
        }, myId.slice(-2).toUpperCase()),
        // Peer avatars
        each(() => session.peers(), p => p.userId, (peer) =>
          h('span', {
            class: 'avatar',
            style: () => `background:${peer.meta?.color ?? '#6366f1'}`,
            title: () => peer.meta?.name ?? peer.userId,
          }, () => peer.userId.slice(-2).toUpperCase())
        ),
      ),
      h('span', { class: 'presence-count text-muted' },
        () => `${peerCount() + 1} user${peerCount() === 0 ? '' : 's'} editing`
      ),
      h('div', { class: 'token-pill' },
        h('span', { class: 'token-dot' }),
        () => session.status(),
      ),
    ),

    // Title input — synced
    h('input', {
      class: 'doc-title',
      value: title,
      oninput: (e) => title.set(e.target.value),
      placeholder: 'Document title...',
    }),

    // Content textarea — synced
    h('textarea', {
      class: 'doc-content',
      value: content,
      oninput: (e) => content.set(e.target.value),
      placeholder: 'Start writing — edits sync to all collaborators in real time...',
      rows: 12,
    }),

    // Word/char count
    h('div', { class: 'doc-meta text-muted' },
      () => `${wordCount()} words · ${charCount()} characters`,
    ),

    // Tags — synced
    h('div', { class: 'tags-section' },
      each(() => tags(), t => t, (tag) =>
        h('span', { class: 'tag tag-new' },
          tag, ' ',
          h('button', {
            onclick: () => tags.set(tags().filter(t => t !== tag)),
          }, '×')
        )
      ),
      h('input', {
        placeholder: 'Add tag...',
        value: newTag,
        oninput: (e) => newTag.set(e.target.value),
        onkeydown: (e) => e.key === 'Enter' && addTag(),
      }),
    ),
  );
}

Decay Signal — Auto-Reset Notifications

createDecaySignal for ephemeral UI state that automatically resets after an idle timeout. Demonstrates toast notifications that dismiss themselves.

import {
  createDecaySignal, signal, computed,
  h, show, each, mountApp,
} from '@nexoraaidrishti/runtime';

// Notification message decays back to null after 3 seconds of no writes
const notification = createDecaySignal<string | null>(null, {
  resetTo: null,
  afterMs: 3000,
  onDecay: () => console.log('Toast dismissed'),
});

// Hover highlight also decays — resets 500ms after mouse leaves
const hoveredId = createDecaySignal<string | null>(null, {
  resetTo: null,
  afterMs: 500,
});

const items = signal([
  { id: 'a', label: 'Save document' },
  { id: 'b', label: 'Publish changes' },
  { id: 'c', label: 'Send invite' },
]);

function notify(msg: string) {
  notification.set(msg); // decay timer resets on each write
}

function DecayApp() {
  return h('div', { class: 'decay-app' },

    // Toast — visible only while notification is non-null
    show(
      () => notification() !== null,
      () => h('div', { class: 'toast toast-success' },
        () => notification()
      ),
    ),

    // Action list — hover highlights decay automatically
    h('ul', { class: 'action-list' },
      each(() => items(), i => i.id, (item) =>
        h('li', {
          class: () => hoveredId() === item.id ? 'action-item highlighted' : 'action-item',
          onmouseenter: () => hoveredId.set(item.id),
          onclick: () => notify(`✓ ${item.label} completed`),
        }, item.label)
      )
    ),
  );
}

mountApp(DecayApp, document.getElementById('app')!);

Security — XSS Guard + Rate Limiting

xssGuardSignal for automatic input sanitization combined with rateLimitSignal to throttle writes. Shows production-grade security primitives.

import {
  xssGuardSignal, rateLimitSignal, RateLimitError,
  signal, h, show, mountApp,
} from '@nexoraaidrishti/runtime';

// User-supplied HTML is stripped of all tags automatically
const comment = xssGuardSignal('', { mode: 'strip' });

// AI inference endpoint: max 5 calls per 10 seconds
const inferenceInput = rateLimitSignal('', {
  maxWrites: 5,
  windowMs: 10_000,
});

const rateLimitError = signal<string | null>(null);

function sendToAI(text: string) {
  try {
    inferenceInput.set(text);
    rateLimitError.set(null);
    // proceed with AI call...
  } catch (e) {
    if (e instanceof RateLimitError) {
      rateLimitError.set(`Rate limited. Retry in ${e.retryAfterMs}ms`);
    }
  }
}

function SecurityApp() {
  return h('div', { class: 'security-demo' },

    // XSS-guarded comment box
    h('div', { class: 'section' },
      h('label', {}, 'Comment (XSS-sanitized):'),
      h('textarea', {
        oninput: (e) => comment.set((e.target as HTMLTextAreaElement).value),
        placeholder: 'Try pasting <script>alert(1)</script>...',
      }),
      h('p', { class: 'sanitized-output' },
        'Stored value: ', () => comment()
      ),
    ),

    // Rate-limited AI input
    h('div', { class: 'section' },
      h('label', {}, 'AI Query (max 5/10s):'),
      h('input', {
        placeholder: 'Type a query...',
        oninput: (e) => sendToAI((e.target as HTMLInputElement).value),
      }),
      show(
        () => rateLimitError() !== null,
        () => h('p', { class: 'error' }, () => rateLimitError()),
      ),
    ),
  );
}

mountApp(SecurityApp, document.getElementById('app')!);

Offline-First Sync

offlineSignal and isOnline for writes that queue to IndexedDB when offline and replay automatically on reconnection. Full offline-first data model.

import {
  offlineSignal, isOnline, onceOnline,
  signal, computed, h, show, each, mountApp,
} from '@nexoraaidrishti/runtime';

// Writes while offline are queued in IndexedDB and replayed on reconnect
const notes = offlineSignal<{ id: string; text: string }[]>('notes', []);
const online = isOnline();         // reactive Signal<boolean>
const pending = notes.pending();  // count of queued offline writes

const draft = signal('');
const noteCount = computed(() => notes().length);

function addNote() {
  const text = draft().trim();
  if (!text) return;
  notes.set([...notes(), { id: Date.now().toString(), text }]);
  draft.set('');
}

// Run a sync once the browser comes back online
onceOnline(() => console.log('Back online — replaying queued writes'));

function OfflineApp() {
  return h('div', { class: 'offline-app' },

    // Connection status banner
    h('div', {
      class: () => online() ? 'status-bar online' : 'status-bar offline',
    },
      () => online()
        ? `Online · ${noteCount()} notes synced`
        : `Offline · ${pending()} writes queued`,
    ),

    // Note input
    h('div', { class: 'input-row' },
      h('input', {
        value: draft,
        oninput: (e) => draft.set((e.target as HTMLInputElement).value),
        onkeydown: (e) => e.key === 'Enter' && addNote(),
        placeholder: 'Write a note (works offline)...',
      }),
      h('button', { class: 'btn btn-primary', onclick: addNote }, 'Add'),
    ),

    // Notes list
    each(() => notes(), n => n.id, (note) =>
      h('div', { class: 'note-item' }, note.text)
    ),
  );
}

mountApp(OfflineApp, document.getElementById('app')!);

Voice Commands

createVoiceSignal maps spoken phrases to signal values via the Web Speech API. Demonstrates hands-free UI control with reactive voice state.

import {
  createVoiceSignal, voiceControl,
  signal, computed, h, show, mountApp,
} from '@nexoraaidrishti/runtime';

// Command signal — spoken phrases map to discrete values
const mode = createVoiceSignal('view', {
  commands: {
    'edit mode': 'edit',
    'view mode': 'view',
    'preview mode': 'preview',
  },
  lang: 'en-US',
});

// Transcription signal — continuous speech-to-text
const searchQuery = signal('');
const voiceHandle = voiceControl(searchQuery, { lang: 'en-US' });

const isListening = computed(() => voiceHandle.listening());
const voiceSupported = mode.supported;

function VoiceApp() {
  return h('div', { class: 'voice-app' },

    show(
      () => !voiceSupported,
      () => h('p', { class: 'warning' }, 'Web Speech API not supported in this browser.'),
    ),

    // Mode indicator
    h('div', { class: 'mode-display' },
      'Current mode: ',
      h('strong', {}, () => mode()),
      h('div', { class: 'voice-hint' }, 'Say "edit mode", "view mode", or "preview mode"'),
    ),

    // Voice search
    h('div', { class: 'voice-search' },
      h('button', {
        class: () => isListening() ? 'btn btn-danger' : 'btn btn-primary',
        onclick: () => isListening() ? voiceHandle.stop() : voiceHandle.start(),
      }, () => isListening() ? '⏹ Stop listening' : '🎤 Start voice search'),
      h('p', { class: 'transcript' },
        'Heard: ', () => searchQuery() || '(nothing yet)'
      ),
    ),

    // Mode controls — also work as regular buttons
    h('div', { class: 'controls' },
      ['view', 'edit', 'preview'].map((m) =>
        h('button', {
          class: () => mode() === m ? 'btn btn-primary' : 'btn btn-ghost',
          onclick: () => mode.set(m),
        }, m)
      ),
    ),
  );
}

mountApp(VoiceApp, document.getElementById('app')!);

Anomaly Detection Dashboard

anomalySignal monitors a time-series metric and flags statistical outliers in real time. Shows z-score detection with a reactive anomaly log.

import {
  anomalySignal, signal, computed,
  h, each, mountApp,
} from '@nexoraaidrishti/runtime';

// Detect anomalies using z-score — flags values > 2.5 std devs from mean
const latencyMs = anomalySignal(0, {
  strategy: 'zscore',
  threshold: 2.5,
  windowSize: 50,
  onAnomaly: (value, meta) => {
    console.warn(`Anomalous latency: ${value}ms (z=${meta.z.toFixed(2)})`);
  },
});

const anomalies = latencyMs.anomalies();  // reactive AnomalyEvent[]
const anomalyCount = computed(() => anomalies().length);
const lastAnomaly  = computed(() => anomalies()[anomalies().length - 1]);

// Simulate incoming telemetry (mostly normal, occasional spike)
setInterval(() => {
  const normal = 50 + Math.random() * 20;
  const spike  = Math.random() < 0.05 ? 300 + Math.random() * 200 : 0;
  latencyMs.set(Math.round(normal + spike));
}, 200);

function AnomalyDashboard() {
  return h('div', { class: 'anomaly-dashboard' },

    h('div', { class: 'metric-row' },
      h('div', { class: 'metric' },
        h('span', { class: 'metric-value' }, () => `${latencyMs()}ms`),
        h('span', { class: 'metric-label' }, 'Current Latency'),
      ),
      h('div', { class: 'metric' },
        h('span', { class: 'metric-value anomaly-count' }, () => anomalyCount()),
        h('span', { class: 'metric-label' }, 'Anomalies Detected'),
      ),
    ),

    h('h3', {}, 'Anomaly Log'),
    h('div', { class: 'anomaly-log' },
      each(
        () => [...anomalies()].reverse().slice(0, 10),
        (a) => a.timestamp,
        (a) => h('div', { class: 'anomaly-row' },
          h('span', { class: 'anomaly-value' }, `${a.value}ms`),
          h('span', { class: 'anomaly-z' }, `z=${a.z.toFixed(2)}`),
          h('span', { class: 'anomaly-time' },
            new Date(a.timestamp).toLocaleTimeString()
          ),
        )
      ),
    ),
  );
}

mountApp(AnomalyDashboard, document.getElementById('app')!);

Dead Signal Detector

createDeadSignalDetector and watchedSignal automatically identify signals that are never read or written after a grace period — keeping codebases clean.

import {
  createDeadSignalDetector, watchedSignal,
  signal, h, each, mountApp,
} from '@nexoraaidrishti/runtime';

// Create detector — no-ops in production by default
const detector = createDeadSignalDetector({
  graceMs: 5_000,        // wait 5s before marking as dead
  checkIntervalMs: 2_000, // scan every 2s
  devOnly: true,          // no-op outside dev mode
  onDead: (label, ageMs) => {
    deadLog.set([...deadLog(), { label, ageMs }]);
  },
});

const deadLog = signal<{ label: string; ageMs: number }[]>([]);

// Active signal — read in the template, stays alive
const activeCount = watchedSignal(0, 'activeCount', detector);

// Orphaned signal — created but never used; will be flagged
const _orphaned = watchedSignal('stale', 'orphanedFeatureFlag', detector);

setInterval(() => activeCount.set(c => c + 1), 1000);

function DeadSignalApp() {
  return h('div', { class: 'dead-signal-app' },

    h('div', { class: 'live-metric' },
      'Active count: ',
      h('strong', {}, () => activeCount()),
    ),

    h('h3', {}, 'Dead Signals Detected'),
    h('p', { class: 'hint' },
      'Wait 5 seconds — "orphanedFeatureFlag" will be flagged as dead.'
    ),
    h('ul', { class: 'dead-log' },
      each(
        () => deadLog(),
        (d) => d.label,
        (d) => h('li', { class: 'dead-entry' },
          h('code', {}, d.label),
          ` — unused for ${(d.ageMs / 1000).toFixed(1)}s`,
        )
      ),
    ),
  );
}

AI Generate + Screen Schema

generate() creates a UI component from a natural-language prompt; captureScreenSchema() then feeds the rendered result back to an LLM for accessibility review.

import {
  generate, captureScreenSchema, schemaToPrompt,
  signal, h, show, mountApp,
} from '@nexoraaidrishti/runtime';

const prompt    = signal('');
const loading   = signal(false);
const generated = signal<HTMLElement | null>(null);
const review    = signal('');
const error     = signal('');

async function generateUI() {
  if (!prompt().trim()) return;
  loading.set(true);
  error.set('');
  try {
    const result = await generate({ prompt: prompt() });
    generated.set(result.element);

    // Capture schema of the generated UI and send to LLM for review
    const schema = captureScreenSchema();
    const schemaText = schemaToPrompt(schema);

    const reviewResult = await fetch('/api/review', {
      method: 'POST',
      body: JSON.stringify({
        system: 'Review this UI for accessibility and UX issues.',
        content: schemaText,
      }),
    });
    const { feedback } = await reviewResult.json();
    review.set(feedback);
  } catch (e) {
    error.set((e as Error).message);
  } finally {
    loading.set(false);
  }
}

function AIGenerateApp() {
  return h('div', { class: 'ai-generate-app' },

    h('div', { class: 'prompt-row' },
      h('input', {
        value: prompt,
        oninput: (e) => prompt.set((e.target as HTMLInputElement).value),
        placeholder: 'Describe a UI component...',
        onkeydown: (e) => e.key === 'Enter' && generateUI(),
      }),
      h('button', {
        class: 'btn btn-primary',
        onclick: generateUI,
        disabled: () => loading(),
      }, () => loading() ? 'Generating…' : 'Generate'),
    ),

    show(() => !!error(),
      () => h('p', { class: 'error' }, () => error())),

    // Generated component preview
    show(() => generated() !== null,
      () => h('div', { class: 'generated-preview' },
        h('h3', {}, 'Generated Component'),
        () => generated()!,
      )),

    // AI accessibility review
    show(() => !!review(),
      () => h('div', { class: 'ai-review' },
        h('h3', {}, 'AI Accessibility Review'),
        h('p', {}, () => review()),
      )),
  );
}

mountApp(AIGenerateApp, document.getElementById('app')!);
