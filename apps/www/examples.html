Example 01
Counter with Time-Travel
Fine-grained signal counter with full time-travel debugging. Step back and forward through every state change.
import {
  signal, computed, label,
  enableTimeTravel, h, each, mountApp
} from '@drishti/runtime';

// Named signals — visible in devtools and time-travel UI
const count   = label('counter', signal(0));
const step    = label('step',    signal(1));
const doubled = computed(() => count() * 2);
const squared = computed(() => count() ** 2);
const isNeg   = computed(() => count() < 0);

// Enable time travel — records every signal mutation
const tt = enableTimeTravel({ maxEntries: 50 });

function increment() { count.set(c => c + step()); }
function decrement() { count.set(c => c - step()); }
function reset()     { count.set(0); step.set(1); }

const Counter = () => h('div', { class: 'counter-app' },
  h('h1', {}, () => `${count()}`),
  h('p', { class: 'muted' },
    () => `doubled: ${doubled()} · squared: ${squared()}`
  ),
  h('div', { class: 'controls' },
    h('button', { onclick: decrement }, '−'),
    h('button', { onclick: increment }, '+'),
    h('button', { onclick: reset, class: 'secondary' }, 'Reset'),
  ),
  h('label', {},
    'Step: ',
    h('input', {
      type: 'number', value: step, min: 1,
      oninput: e => step.set(Number(e.target.value)),
    }),
  ),

  // Time-travel controls
  h('div', { class: 'tt-controls' },
    h('button', {
      onclick: () => tt.back(),
      disabled: () => tt.canGoBack() === false
    }, '← Undo'),
    h('button', {
      onclick: () => tt.forward(),
      disabled: () => tt.canGoForward() === false
    }, 'Redo →'),
    h('button', { onclick: () => tt.replay({ speed: 2 }) }, 'Replay ×2'),
  ),

  // Live history
  h('ol', { class: 'tt-history', reversed: true },
    each(tt.entries, (e, i) =>
      h('li', { key: e.id },
        h('span', { class: 'tt-label' }, e.label),
        h('span', { class: 'tt-value' }, `${JSON.stringify(e.prev)} → ${JSON.stringify(e.next)}`),
        h('button', { onclick: () => tt.restore(e) }, 'Restore'),
      )
    )
  ),
);

mountApp(Counter, document.getElementById('app'));
Example 02
TodoMVC
Full todo app with add, complete, filter, and clear. Demonstrates signal arrays, computed filtering, and keyed list rendering.
import {
  signal, computed, batch,
  h, each, show, mountApp
} from '@drishti/runtime';

type Todo = { id: number; text: string; done: boolean };
type Filter = 'all' | 'active' | 'completed';

const todos   = signal<Todo[]>([]);
const filter  = signal<Filter>('all');
const input   = signal('');
let   nextId  = 1;

const filtered = computed(() => {
  const f = filter();
  return todos().filter(t =>
    f === 'all'       ? true  :
    f === 'active'    ? !t.done :
    f === 'completed' ?  t.done : true
  );
});

const remaining  = computed(() => todos().filter(t => !t.done).length);
const allDone    = computed(() => todos().length > 0 && remaining() === 0);

function add() {
  const text = input().trim();
  if (!text) return;
  todos.set([...todos(), { id: nextId++, text, done: false }]);
  input.set('');
}

function toggle(id: number) {
  todos.set(todos().map(t => t.id === id ? { ...t, done: !t.done } : t));
}

function remove(id: number) {
  todos.set(todos().filter(t => t.id !== id));
}

function toggleAll() {
  const done = !allDone();
  todos.set(todos().map(t => ({ ...t, done })));
}

function clearDone() {
  todos.set(todos().filter(t => !t.done));
}

const TodoApp = () => h('div', { class: 'todo-app' },
  h('h1', {}, 'todos'),

  // Input row
  h('div', { class: 'todo-input-row' },
    h('input', {
      class: 'todo-checkbox-all', type: 'checkbox',
      checked: allDone, onchange: toggleAll
    }),
    h('input', {
      class: 'todo-input', placeholder: 'What needs to be done?',
      value: input,
      oninput: (e) => input.set((e.target as HTMLInputElement).value),
      onkeydown: (e) => (e as KeyboardEvent).key === 'Enter' && add(),
      autofocus: true,
    }),
  ),

  // Todo list
  h('ul', { class: 'todo-list' },
    each(filtered, (todo) =>
      h('li', {
        key: todo.id,
        class: () => todo.done ? 'todo-item done' : 'todo-item'
      },
        h('input', {
          type: 'checkbox', checked: todo.done,
          onchange: () => toggle(todo.id),
        }),
        h('span', { class: 'todo-text' }, todo.text),
        h('button', {
          class: 'todo-delete', onclick: () => remove(todo.id)
        }, '×'),
      )
    )
  ),

  // Footer
  show(() => todos().length > 0,
    h('footer', { class: 'todo-footer' },
      h('span', {}, () => `${remaining()} item${remaining() === 1 ? '' : 's'} left`),
      h('div', { class: 'filters' },
        ...((['all', 'active', 'completed'] as Filter[]).map(f =>
          h('button', {
            class: () => filter() === f ? 'filter-btn active' : 'filter-btn',
            onclick: () => filter.set(f),
          }, f)
        )),
      ),
      h('button', {
        class: 'clear-btn',
        onclick: clearDone,
        style: () => todos().some(t => t.done) ? '' : 'visibility:hidden',
      }, 'Clear completed'),
    )
  ),
);

mountApp(TodoApp, document.getElementById('app'));
Example 03
Data Fetching
Reactive data loading with createResource — automatic refetch on dependency change, loading skeletons, and error handling.
import {
  signal, createResource,
  h, show, each, mountApp
} from '@drishti/runtime';

interface Post {
  id: number; title: string; body: string; userId: number;
}

const userId = signal(1);

// createResource re-fetches automatically when userId changes
const [posts, { refetch }] = createResource(
  userId,
  async (id) => {
    const res = await fetch(`https://jsonplaceholder.typicode.com/posts?userId=${id}`);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return res.json() as Promise<Post[]>;
  }
);

const PostList = () => h('div', { class: 'post-list' },

  // User selector
  h('div', { class: 'user-selector' },
    h('label', {}, 'User ID: '),
    h('select', {
      value: userId,
      onchange: (e) => userId.set(Number((e.target as HTMLSelectElement).value))
    },
      ...[1,2,3,4,5].map(n =>
        h('option', { value: n }, `User ${n}`)
      )
    ),
    h('button', { onclick: refetch }, '↻ Refresh'),
  ),

  // Loading skeleton
  show(() => posts.loading(),
    h('div', { class: 'skeleton-list' },
      ...Array(3).fill(null).map(() =>
        h('div', { class: 'skeleton-card' })
      )
    )
  ),

  // Error state
  show(() => !!posts.error(),
    h('div', { class: 'error-banner' },
      h('p', {}, () => `Error: ${posts.error()?.message}`),
      h('button', { onclick: refetch }, 'Retry'),
    )
  ),

  // Posts grid
  show(() => !posts.loading() && !posts.error(),
    h('div', { class: 'posts-grid' },
      each(() => posts() ?? [], (post) =>
        h('article', { key: post.id, class: 'post-card' },
          h('h3', {}, post.title),
          h('p', {}, post.body),
          h('span', { class: 'post-id' }, `#${post.id}`),
        )
      )
    )
  ),
);

mountApp(PostList, document.getElementById('app'));
Example 04
Form Validation
Reactive form with per-field validation, async username availability check, and submit handling via createForm.
import { createForm, h, show, mountApp } from '@drishti/runtime';

const form = createForm({
  username: {
    initial: '',
    validators: [
      (v) => v.length < 3 ? 'Min 3 characters' : null,
      (v) => /[^a-z0-9_]/.test(v) ? 'Only a-z, 0-9, _' : null,
      async (v) => {
        const res = await fetch(`/api/check-username?q=${v}`).then(r => r.json());
        return res.taken ? 'Username taken' : null;
      }
    ],
  },
  email: {
    initial: '',
    validators: [
      (v) => !v.includes('@') ? 'Invalid email' : null,
    ],
  },
  password: {
    initial: '',
    validators: [
      (v) => v.length < 8  ? 'Min 8 characters'    : null,
      (v) => !/[A-Z]/.test(v) ? 'Needs uppercase letter' : null,
      (v) => !/[0-9]/.test(v) ? 'Needs a number'         : null,
    ],
  },
});

function Field({ label, name, type = 'text' }: { label: string; name: keyof typeof form.fields; type?: string }) {
  const field = form.fields[name];
  return h('div', { class: 'field-group' },
    h('label', {}, label),
    h('input', {
      type,
      value: field.value,
      class: () => field.error() && field.touched() ? 'input-error' : '',
      oninput: (e) => field.set((e.target as HTMLInputElement).value),
      onblur: () => field.touch(),
    }),
    show(() => !!field.error() && field.touched(),
      h('span', { class: 'field-error' }, () => field.error() ?? '')
    ),
  );
}

const RegisterForm = () => h('form', {
  class: 'register-form',
  onsubmit: form.submit(async (values) => {
    await fetch('/api/register', {
      method: 'POST',
      body: JSON.stringify(values),
      headers: { 'Content-Type': 'application/json' },
    });
  }),
},
  h('h2', {}, 'Create Account'),
  Field({ label: 'Username', name: 'username' }),
  Field({ label: 'Email',    name: 'email',    type: 'email' }),
  Field({ label: 'Password', name: 'password', type: 'password' }),
  h('button', {
    type: 'submit',
    disabled: () => !form.valid() || form.submitting(),
    class: 'btn-primary',
  }, () => form.submitting() ? 'Creating...' : 'Create Account'),
  show(() => form.submitted(),
    h('p', { class: 'success' }, 'Account created!')
  ),
);

mountApp(RegisterForm, document.getElementById('app'));
Example 05
State Machine Wizard
Multi-step onboarding wizard driven by a typed state machine. Guards prevent invalid transitions; progress is tracked reactively.
import { createMachine, signal, h, show, mountApp } from '@drishti/runtime';

const name    = signal('');
const email   = signal('');
const plan    = signal('');
const card    = signal('');
const submitting = signal(false);

const wizard = createMachine({
  initial: 'personal',
  states: {
    personal: {
      on: {
        NEXT: {
          target: 'plan',
          guard: () => name().length > 1 && email().includes('@'),
        },
      },
      meta: { title: 'Personal Info', step: 1 },
    },
    plan: {
      on: {
        NEXT: { target: 'payment', guard: () => !!plan() },
        BACK: 'personal',
      },
      meta: { title: 'Choose Plan', step: 2 },
    },
    payment: {
      on: {
        SUBMIT: { target: 'processing', guard: () => card().length === 16 },
        BACK: 'plan',
      },
      meta: { title: 'Payment', step: 3 },
    },
    processing: {
      entry: async () => {
        submitting.set(true);
        await new Promise(r => setTimeout(r, 2000));
        wizard.send('DONE');
      },
      on: { DONE: 'complete' },
      meta: { title: 'Processing…', step: 3 },
    },
    complete: {
      meta: { title: 'All Done!', step: 4 },
    },
  },
});

const Wizard = () => h('div', { class: 'wizard' },

  // Progress bar
  h('div', { class: 'wizard-progress' },
    h('div', {
      class: 'wizard-progress-bar',
      style: () => `width: ${(wizard.state().meta.step / 4) * 100}%`
    })
  ),
  h('h2', {}, () => wizard.state().meta.title),

  // Step 1: Personal
  show(() => wizard.matches('personal'),
    h('div', { class: 'step' },
      h('input', { placeholder: 'Full name', value: name, oninput: e => name.set(e.target.value) }),
      h('input', { type: 'email', placeholder: 'Email', value: email, oninput: e => email.set(e.target.value) }),
      h('button', { onclick: () => wizard.send('NEXT') }, 'Continue →'),
    )
  ),

  // Step 2: Plan
  show(() => wizard.matches('plan'),
    h('div', { class: 'step' },
      ...['free', 'pro', 'enterprise'].map(p =>
        h('button', {
          class: () => plan() === p ? 'plan-btn active' : 'plan-btn',
          onclick: () => plan.set(p),
        }, p.charAt(0).toUpperCase() + p.slice(1))
      ),
      h('div', { class: 'step-nav' },
        h('button', { onclick: () => wizard.send('BACK') }, '← Back'),
        h('button', { onclick: () => wizard.send('NEXT') }, 'Continue →'),
      ),
    )
  ),

  // Step 3: Payment
  show(() => wizard.matches('payment'),
    h('div', { class: 'step' },
      h('input', { placeholder: 'Card number (16 digits)', maxlength: 16, value: card, oninput: e => card.set(e.target.value) }),
      h('div', { class: 'step-nav' },
        h('button', { onclick: () => wizard.send('BACK') }, '← Back'),
        h('button', { onclick: () => wizard.send('SUBMIT') }, 'Pay Now'),
      ),
    )
  ),

  // Processing
  show(() => wizard.matches('processing'),
    h('div', { class: 'step processing' },
      h('div', { class: 'spinner' }),
      h('p', {}, 'Processing your payment...'),
    )
  ),

  // Complete
  show(() => wizard.matches('complete'),
    h('div', { class: 'step complete' },
      h('p', { class: 'success-icon' }, '✓'),
      h('h3', {}, () => `Welcome, ${name()}!`),
      h('p', {}, () => `You're on the ${plan()} plan.`),
    )
  ),
);

mountApp(Wizard, document.getElementById('app'));
Example 06
LLM Chat
Full chat UI with streaming responses, extended thinking visualization, tool call display, and message history using createLLMStream.
import {
  signal, computed,
  createLLMStream, createThinkingState, createToolCallState,
  h, show, each, mountApp
} from '@drishti/runtime';

type Message = { role: 'user' | 'assistant'; content: string; ts: number };

const messages  = signal<Message[]>([]);
const input     = signal('');
const showThink = signal(false);

// Create streaming connection
const stream  = createLLMStream({ endpoint: '/api/chat' });
const thinking = createThinkingState();
const tools   = createToolCallState();

const isStreaming = computed(() => stream.status() === 'streaming');

async function send() {
  const text = input().trim();
  if (!text || isStreaming()) return;

  // Append user message
  messages.set([...messages(), { role: 'user', content: text, ts: Date.now() }]);
  input.set('');

  // Stream response
  await stream.send({
    messages: messages().map((m) => ({ role: m.role, content: m.content })),
  });

  // Commit streamed response to history
  messages.set([...messages(), { role: 'assistant', content: stream.text(), ts: Date.now() }]);
}

const ChatApp = () => h('div', { class: 'chat-app' },

  // Messages
  h('div', { class: 'messages' },
    each(messages, (msg) =>
      h('div', { key: msg.ts, class: `message ${msg.role}` },
        h('div', { class: 'message-role' }, msg.role),
        h('div', { class: 'message-content' }, msg.content),
      )
    ),

    // Live streaming token
    show(isStreaming,
      h('div', { class: 'message assistant streaming' },
        show(() => thinking.active() && showThink(),
          h('div', { class: 'thinking-block' },
            h('div', { class: 'thinking-label' }, 'Thinking…'),
            h('p', { class: 'thinking-text' }, () => thinking.text()),
          )
        ),
        h('div', { class: 'message-content' }, () => stream.text()),
        h('span', { class: 'cursor' }),
      )
    ),
  ),

  // Tool calls badge
  show(() => tools.active().length > 0,
    h('div', { class: 'tool-calls' },
      each(tools.active, (tc) =>
        h('div', { key: tc.id, class: 'tool-call' },
          h('span', { class: 'tool-name' }, tc.name),
          h('span', { class: 'tool-status' }, tc.status),
        )
      )
    )
  ),

  // Input bar
  h('div', { class: 'chat-input-bar' },
    h('textarea', {
      placeholder: 'Message AI...',
      value: input,
      oninput: (e) => input.set((e.target as HTMLTextAreaElement).value),
      onkeydown: (e) => {
        if ((e as KeyboardEvent).key === 'Enter' && !(e as KeyboardEvent).shiftKey) {
          e.preventDefault();
          send();
        }
      },
    }),
    h('div', { class: 'chat-options' },
      h('label', {},
        h('input', { type: 'checkbox', checked: showThink, onchange: () => showThink.set(!showThink()) }),
        ' Show thinking',
      ),
      h('span', { class: 'token-count' }, () => `${stream.tokens()} tokens`),
    ),
    h('button', {
      onclick: send,
      disabled: isStreaming,
      class: 'send-btn',
    }, () => isStreaming() ? 'Streaming...' : 'Send'),
  ),
);

mountApp(ChatApp, document.getElementById('app'));

Feature Flags & A/B Testing
Multi-armed bandit A/B testing with auto-rollout, percentage-based flags, and real-time variant switching.
import {
  createFlag, createFlagStore, createBandit, createAutoRollout,
  signal, h, show, each, mountApp
} from '@drishti/runtime';

const userId = signal('user-abc-123');

// Simple boolean flag with percentage rollout
const darkMode = createFlag({
  key: 'dark-mode-v2',
  defaultValue: false,
  rules: [{ percentage: 50, value: true }],
  userId: userId,
});

// Multi-variant flag
const checkoutVariant = createFlag({
  key: 'checkout-flow',
  defaultValue: 'control',
  variants: [
    { key: 'control',    weight: 40 },
    { key: 'one-page',   weight: 30 },
    { key: 'progressive', weight: 30 },
  ],
  userId: userId,
});

// Bandit — converges automatically on best-performing variant
const bandit = createBandit({
  key: 'cta-button',
  arms: [
    { key: 'Get Started',  initialReward: 0.1 },
    { key: 'Start for Free', initialReward: 0.12 },
    { key: 'Try DRISHTI',   initialReward: 0.08 },
  ],
  algorithm: 'thompson',
});

// Auto-rollout — ramps 0→100% over 7 days while error rate < 1%
const newEditor = createAutoRollout({
  key: 'new-editor',
  targetPercentage: 100,
  durationDays: 7,
  gates: [
    { metric: 'error_rate', threshold: 0.01, direction: 'below' },
    { metric: 'p95_latency_ms', threshold: 200, direction: 'below' },
  ],
});

const FlagDemo = () => h('div', { class: 'flag-demo' },
  h('h2', {}, 'Feature Flags'),

  h('div', { class: 'flag-table' },
    h('div', { class: 'flag-row' },
      h('span', {}, 'dark-mode-v2'),
      h('span', { class: () => darkMode.value() ? 'badge-green' : 'badge' },
        () => String(darkMode.value())
      ),
    ),
    h('div', { class: 'flag-row' },
      h('span', {}, 'checkout-flow'),
      h('span', { class: 'badge' }, () => checkoutVariant.value()),
    ),
    h('div', { class: 'flag-row' },
      h('span', {}, 'CTA Bandit winner'),
      h('span', { class: 'badge' }, () => bandit.bestArm()?.key ?? '—'),
    ),
    h('div', { class: 'flag-row' },
      h('span', {}, 'new-editor rollout'),
      h('span', {}, () => `${newEditor.percentage().toFixed(1)}%`),
    ),
  ),

  // Dynamic CTA from bandit
  h('button', {
    class: 'btn-primary btn-lg',
    onclick: () => bandit.reward(bandit.value(), 1),
  }, () => bandit.value()),

  // Override user id for testing
  h('input', {
    placeholder: 'Override user ID',
    value: userId,
    oninput: (e) => userId.set((e.target as HTMLInputElement).value),
  }),
);

mountApp(FlagDemo, document.getElementById('app'));
Example 08
Real-time Sync
Collaborative whiteboard using createCollabSession for CRDT state and connectWS for cursor presence. All conflict-free.
import {
  createCollabSession, connectWS, signal, computed, effect,
  h, each, show, mountApp
} from '@drishti/runtime';

type Point = { x: number; y: number };
type Shape = { id: string; type: 'rect' | 'circle'; pos: Point; color: string };

const myId = signal(crypto.randomUUID());
const myColor = signal(#6366f1');

// CRDT session — all mutations are conflict-free
const session = createCollabSession({
  room: 'whiteboard-1',
  transport: 'wss://collab.example.com',
  userId: myId(),
  presence: { color: myColor(), name: 'Me' },
});

// Shared signals — sync automatically across all clients
const shapes = session.signal<Shape[]>('shapes', []);
const title  = session.signal<string>('title', 'Untitled Board');

// Local cursor position (not shared via CRDT — use presence instead)
const cursor = signal<Point>({ x: 0, y: 0 });

// Computed: number of online collaborators
const onlineCount = computed(() => session.presence().length);

function addShape(type: 'rect' | 'circle') {
  const shape: Shape = {
    id: crypto.randomUUID(),
    type,
    pos: { x: cursor().x, y: cursor().y },
    color: myColor(),
  };
  shapes.set([...shapes(), shape]);
  // Change syncs to all clients instantly via WebSocket + CRDT
}

function deleteShape(id: string) {
  shapes.set(shapes().filter(s => s.id !== id));
}

const Whiteboard = () => h('div', { class: 'whiteboard' },

  // Toolbar
  h('div', { class: 'wb-toolbar' },
    h('input', {
      value: title,
      class: 'wb-title',
      oninput: (e) => title.set((e.target as HTMLInputElement).value),
    }),
    h('button', { onclick: () => addShape('rect') }, '+ Rect'),
    h('button', { onclick: () => addShape('circle') }, '+ Circle'),
    h('input', { type: 'color', value: myColor, oninput: (e) => myColor.set((e.target as HTMLInputElement).value) }),
    h('span', { class: 'presence-badge' }, () => `${onlineCount()} online`),
  ),

  // Canvas
  h('div', {
    class: 'canvas',
    onmousemove: (e) => {
      const rect = (e.currentTarget as HTMLDivElement).getBoundingClientRect();
      cursor.set({ x: e.clientX - rect.left, y: e.clientY - rect.top });
      // Broadcast cursor position to presence
      session.updatePresence({ cursor: cursor() });
    },
  },
    // Render shapes
    each(shapes, (shape) =>
      h('div', {
        key: shape.id,
        class: `shape ${shape.type}`,
        style: () => `left: ${shape.pos.x}px; top: ${shape.pos.y}px; background: ${shape.color}`,
        ondblclick: () => deleteShape(shape.id),
      })
    ),

    // Render remote cursors
    each(session.presence, (peer) =>
      show(() => peer.cursor && peer.userId !== myId(),
        h('div', {
          key: peer.userId,
          class: 'remote-cursor',
          style: () => `left: ${peer.cursor.x}px; top: ${peer.cursor.y}px; border-color: ${peer.color}`,
        },
          h('span', { class: 'cursor-label', style: () => `background: ${peer.color}` }, peer.name),
        )
      )
    ),
  ),

  // Status bar
  h('div', { class: 'wb-status' },
    h('span', {}, () => `${shapes().length} shapes`),
    h('span', { class: () => `conn-status ${session.status()}` },
      () => session.status()
    ),
  ),
);

mountApp(Whiteboard, document.getElementById('app'));
