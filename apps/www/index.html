Quick Start

Intuitive by Design

From a simple counter to real-time AI streaming and collaborative editing — DRISHTI handles it all with the same primitives.

import { signal, computed, h, mountApp } from '@drishti/runtime';

const count = signal(0);
const doubled = computed(() => count() * 2);

function Counter() {
  return h('div', { class: 'counter' },
    h('button', { onclick: () => count.set(count() - 1) }, '-'),
    () => `${count()} (×2 = ${doubled()})`,
    h('button', { onclick: () => count.set(count() + 1) }, '+'),
  );
}

mountApp(Counter, document.getElementById('app')!);
import { signal, computed, h, each, show, mountApp } from '@drishti/runtime';

interface Todo { id: number; text: string; done: boolean; }

const todos = signal<Todo[]>([]);
const input = signal('');
const remaining = computed(() => todos().filter(t => !t.done).length);

function addTodo() {
  if (!input().trim()) return;
  todos.set([...todos(), { id: Date.now(), text: input(), done: false }]);
  input.set('');
}

function App() {
  return h('div', { class: 'app' },
    h('input', { value: input, oninput: (e) => input.set(e.target.value), placeholder: 'New todo…' }),
    h('button', { onclick: addTodo }, 'Add'),
    each(() => todos(), t => t.id, t =>
      h('li', { onclick: () => todos.set(todos().map(x => x.id === t.id ? {...x, done: !x.done} : x)) },
        () => t.text
      )
    ),
    () => `${remaining()} remaining`,
  );
}
import { signal, h, mountApp, createLLMStream } from '@drishti/runtime';

const response = signal('');
const thinking = signal(false);

const stream = createLLMStream({
  onChunk: (text) => response.set(response() + text),
  onStart: () => { thinking.set(true); response.set(''); },
  onDone:  () => thinking.set(false),
});

function ChatApp() {
  const input = signal('');
  return h('div', {},
    h('input', { value: input, oninput: (e) => input.set(e.target.value) }),
    h('button', { onclick: () => stream.start({ messages: [{ role: 'user', content: input() }] }) }, 'Ask AI'),
    () => thinking() ? '⏳ Thinking…' : '',
    () => response(),
  );
}
import { createMachine, h, mountApp } from '@drishti/runtime';

const traffic = createMachine({
  initial: 'red',
  states: {
    red:    { on: { NEXT: 'green'  } },
    green:  { on: { NEXT: 'yellow' } },
    yellow: { on: { NEXT: 'red'    } },
  },
});

function TrafficLight() {
  return h('div', { class: 'light-container' },
    h('div', { class: () => `light light-${traffic.state()}` }),
    h('p', {}, () => `State: ${traffic.state()}`),
    h('button', { onclick: () => traffic.send('NEXT') }, 'Next State'),
  );
}

mountApp(TrafficLight, document.getElementById('app')!);
import { signal, createCollabSession, h, mountApp } from '@drishti/runtime';

const session = createCollabSession({
  userId: 'user-' + Math.random().toString(36).slice(2),
  wsUrl:  'wss://your-server/collab',
});

// Any signal becomes collaborative
const sharedDoc = signal('');
session.sync('document', sharedDoc);

function CollabEditor() {
  return h('div', {},
    h('p', {}, () => `${session.peers().length + 1} users editing`),
    h('textarea', {
      value: sharedDoc,
      oninput: (e) => sharedDoc.set(e.target.value),
      placeholder: 'Start typing — updates sync to all users in real-time…',
    }),
  );
}

mountApp(CollabEditor, document.getElementById('app')!);

AI-Native

Built for the
Age of AI Agents

DRISHTI is the first framework designed from the ground up for AI agent interaction. Give any AI assistant full observability and direct write access to your entire application state.

No DOM scraping. No heuristics. Pure reactive signals with a clean read/write API — exactly how an AI agent needs to operate.

Read the entire app state with a single call to _getDevSignals()
Write to any signal directly — UI updates instantly, no REST API needed
Observe every state change in real time with _setSignalWriteHook
Understand user intent with createIntentEngine()
Stream LLM responses directly into reactive signals with createLLMStream()
// Give an AI agent full read/write access to your app state
import { _setSignalWriteHook, _getDevSignals, createIntentEngine } from '@drishti/runtime';

// Read the full app state at any time
const state = _getDevSignals();
// → { count: Signal<0>, filter: Signal<'all'>, todos: Signal<[...]>, ... }

// AI writes directly to any signal
state['filter'].set('completed');   // UI updates instantly — no REST API needed
state['count'].set(42);

// Observe every state change
_setSignalWriteHook((label, prev, next) => {
  aiAgent.observe({ label, prev, next, timestamp: Date.now() });
});

// Intent engine: AI understands user actions and controls UI
const intent = createIntentEngine({
  onIntent: ({ action, params }) => {
    if (action === 'filter')   filterSignal.set(params.query);
    if (action === 'navigate') router.push(params.path);
    if (action === 'search')   searchSignal.set(params.text);
  }
});
