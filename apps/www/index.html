Quick Examples

Intuitive by Design

From simple counters to AI streaming and realtime collaboration — DRISHTI handles it all.

import { signal, computed, h, mountApp } from '@drishti/runtime';

// Create reactive signals
const count = signal(0);
const doubled = computed(() => count() * 2);
const isEven = computed(() => count() % 2 === 0);

// Build UI with h() — direct DOM, no vdom
const App = () => h('div', { class: 'counter' },
  h('p', {}, () => `Count: ${count()} (doubled: ${doubled()})`),
  h('p', { class: () => isEven() ? 'even' : 'odd' }, () => isEven() ? 'Even' : 'Odd'),
  h('button', { onclick: () => count.set(count() - 1) }, '−'),
  h('button', { onclick: () => count.set(count() + 1) }, '+'),
);

mountApp(App, document.getElementById('app'));
import { signal, computed, h, each, show, mountApp } from '@drishti/runtime';

const todos = signal([]);
const filter = signal('all');
const newText = signal('');

const filtered = computed(() => {
  const f = filter();
  return todos().filter(t =>
    f === 'all' ? true :
    f === 'done' ? t.done : !t.done
  );
});

function addTodo() {
  const text = newText().trim();
  if (!text) return;
  todos.set([...todos(), { id: Date.now(), text, done: false }]);
  newText.set('');
}

const App = () => h('div', {},
  h('input', {
    value: newText,
    oninput: e => newText.set(e.target.value),
    onkeydown: e => e.key === 'Enter' && addTodo(),
    placeholder: 'Add a todo...'
  }),
  h('button', { onclick: addTodo }, 'Add'),
  each(filtered, (todo) =>
    h('li', { key: todo.id },
      h('input', { type: 'checkbox', checked: todo.done,
        onchange: () => todos.set(todos().map(t =>
          t.id === todo.id ? { ...t, done: !t.done } : t)) }),
      h('span', {}, todo.text),
    )
  ),
);

mountApp(App, document.getElementById('app'));
import { createLLMStream, createThinkingState, h, show } from '@drishti/runtime';

// Create a streaming LLM connection
const stream = createLLMStream({
  endpoint: '/api/chat',
  model: 'claude-3-5-sonnet',
});

// Optional extended thinking visualization
const thinking = createThinkingState();

// Reactive signals updated as tokens arrive
// stream.text()   — current streamed text
// stream.status() — 'idle' | 'streaming' | 'done' | 'error'
// stream.tokens() — token count

async function send(message) {
  await stream.send({ messages: [{ role: 'user', content: message }] });
}

const ChatUI = () => h('div', { class: 'chat' },
  show(() => stream.status() === 'streaming',
    h('div', { class: 'thinking' }, 'AI is thinking...')
  ),
  h('div', { class: 'response' }, () => stream.text()),
  h('p', { class: 'tokens' }, () => `Tokens: ${stream.tokens()}`),
  h('button', {
    onclick: () => send('Explain fine-grained reactivity'),
    disabled: () => stream.status() === 'streaming'
  }, 'Ask AI'),
);
import { createMachine, h, show } from '@drishti/runtime';

// XState-inspired typed state machine
const traffic = createMachine({
  initial: 'red',
  states: {
    red:    { on: { NEXT: 'green'  }, meta: { label: 'Stop',  color: '#ef4444' } },
    green:  { on: { NEXT: 'yellow' }, meta: { label: 'Go',    color: '#10b981' } },
    yellow: { on: { NEXT: 'red'    }, meta: { label: 'Slow',  color: '#f59e0b' } },
  },
});

// Reactive: traffic.state() updates the UI automatically
const TrafficLight = () => h('div', {},
  h('div', {
    class: 'light',
    style: () => `background: ${traffic.state().meta.color}`
  }),
  h('p', {}, () => traffic.state().meta.label),
  h('button', {
    onclick: () => traffic.send('NEXT')
  }, 'Next Phase'),

  // Show history with time-travel
  h('p', {}, () => `Transitions: ${traffic.history().length}`),
);
import { createCollabSession, signal, h } from '@drishti/runtime';

// CRDT-backed collaborative session with vector clocks
const session = createCollabSession({
  room: 'doc-42',
  transport: 'wss://collab.example.com',
  userId: 'user-alice',
});

// Shared signals — conflict-free across all clients
const title = session.signal('title', 'Untitled Document');
const content = session.signal('content', '');

// Presence — see who's online
// session.presence() → [{ userId, cursor, color, name }]

const CollabDoc = () => h('div', { class: 'doc' },
  h('div', { class: 'presence' },
    () => `${session.presence().length} collaborators online`
  ),
  h('input', {
    value: title,
    oninput: e => title.set(e.target.value),
    placeholder: 'Document title...'
  }),
  h('textarea', {
    value: content,
    oninput: e => content.set(e.target.value),
  }),
  h('span', { class: 'status' }, () => session.status()),
);
Why DRISHTI

Everything You Need

One framework, zero tradeoffs. Production-ready from day one.

⚡

Fine-Grained Signals

No virtual DOM diffing. Each signal update patches only the exact DOM nodes that depend on it. Sub-millisecond updates at any scale.

🤖

AI-Native API

_setSignalWriteHook, _getDevSignals, and createIntentEngine give AI agents complete observability and control of your app state.

⏱️

Time-Travel Debugging

enableTimeTravel() records every signal mutation. Snapshot, restore, and replay entire application states — perfect for debugging and testing.

🔄

Real-Time Collaboration

Built-in CRDT with vector clocks. createCollabSession delivers conflict-free shared state across any number of clients over WebSockets.

🧠

State Machines

XState-inspired typed machines via createMachine. Define states, transitions, guards, and actions — fully reactive with signal integration.

🎨

Figma → Code

Export designs as .dr single-file components directly from Figma. Design tokens, genome palettes, and emotion-adaptive theming built in.

🏝️

Islands Architecture

defineIsland and hydrateIslands support 4 hydration strategies: eager, lazy, viewport, and idle — ship minimal JS to the client.

🔒

Production Security

HIPAA mode, CSP generation, GDPR consent management, and Zero-Trust mesh. sanitizeHTML, sanitizeURL, and audit logging built in.

📦

Deploy Anywhere

First-class adapters for Vercel, Netlify, Cloudflare Workers, and edge runtimes. Generate config with a single function call.

AI-Native

Built for the
Age of AI Agents

DRISHTI is the first framework designed from the ground up for AI agent interaction. Give your AI assistant full observability and control of your entire application state.

No scraping the DOM. No heuristics. Pure reactive signals with direct read/write access — exactly how an AI agent needs to work.

Read the entire app state with a single call to _getDevSignals()
Write to any signal directly — UI updates instantly
Observe every state change in real time with _setSignalWriteHook
Understand user intent with createIntentEngine
Stream LLM responses directly into reactive signals
// AI agent gets full app state snapshot
const snapshot = _getDevSignals();
// { count: 0, filter: 'all', todos: [...], ... }

// AI writes directly to any signal
snapshot['filter'].set('completed');
// → UI updates instantly, no rerender needed

// Observe every state change in real time
_setSignalWriteHook((label, prev, next) => {
  aiAgent.observe({ label, prev, next, ts: Date.now() });
});

// Understand what the user is trying to do
const intent = createIntentEngine({
  signals: _getDevSignals(),
  onIntent: (intent) => aiAgent.suggest(intent),
});

// Time-travel: let AI replay any past state
const tt = enableTimeTravel();
tt.restore(tt.entries()[0]);
// → app jumps back to initial state
